DR. HAIAT GENERAL HOSPITAL - HOSPITAL MANAGEMENT SYSTEM (HMS)
COMPREHENSIVE PLANNING DOCUMENT
================================================================================

PROJECT OVERVIEW
================================================================================
Hospital Name: Dr. Haiat General Hospital
Domain: drhaiatgh.com
HMS Console: console.drhaiatgh.com
API Endpoint: api.drhaiatgh.com

Hospital Type: Small-scale Gynecology focused hospital
Capacity: 
  - 1 Operation Theatre (OT)
  - 1 Post-OT room
  - 1 Laboratory
  - 1 USG room
  - 3 Private Cabins
  - 1 Ward (5 beds)
  - 1 Doctor Chamber

Staff:
  - 4 Doctors (2 Gynecology Surgeons + 2 Others)
  - Multiple support staff (nurses, manager, receptionist, lab assistant, guards)
  - 1 Medicine store room (internal use, not commercial pharmacy)

================================================================================
TECHNOLOGY STACK
================================================================================

Frontend (React + TypeScript):
  - React 18.x
  - TypeScript 5.x
  - Vite (Build tool)
  - React Router v6
  - TanStack Query (React Query) v5
  - Zustand (Global state)
  - React Hook Form + Zod
  - Tailwind CSS + Headless UI
  - date-fns
  - recharts (Dashboard charts)
  - react-hot-toast (Notifications)
  - react-to-print (Invoice printing)
  - axios

Backend (NestJS + TypeScript):
  - NestJS 10.x
  - TypeScript 5.x
  - Node.js 20 LTS
  - Prisma ORM
  - PostgreSQL 15
  - Redis 7 (Cache & Queue)
  - Passport.js (JWT Authentication)
  - Bull (Queue management)
  - @nestjs/swagger (API Documentation)
  - class-validator, class-transformer
  - bcrypt (Password hashing)
  - node-cron (Scheduled tasks)

DevOps & Tools:
  - Git & GitHub
  - PM2 (Process management)
  - Let's Encrypt (SSL)
  - Nginx (Reverse proxy)

import { Controller, Post, Body, UseGuards, Request } from '@nestjs/common';
import { AuthService } from './auth.service';
import { LoginDto } from './dto/login.dto';
import { JwtAuthGuard } from './guards/jwt-auth.guard';
import { ApiTags, ApiOperation, ApiBearerAuth } from '@nestjs/swagger';

@ApiTags('Authentication')
@Controller('api/v1/auth')
export class AuthController {
  constructor(private authService: AuthService) {}

  @Post('login')
  @ApiOperation({ summary: 'User login' })
  async login(@Body() loginDto: LoginDto) {
    return this.authService.login(loginDto);
  }

  @Post('logout')
  @UseGuards(JwtAuthGuard)
  @ApiBearerAuth()

    return { message: 'Logged out successfully' };
  }

  @Post('me')
  @UseGuards(JwtAuthGuard)
  @ApiBearerAuth()
  @ApiOperation({ summary: 'Get current user' })
  async me(@Request() req) {

  }
}

// src/modules/auth/auth.service.ts
import { Injectable, UnauthorizedException } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { PrismaService } from '../../prisma/prisma.service';
import * as bcrypt from 'bcrypt';
import { LoginDto } from './dto/login.dto';

@Injectable()
export class AuthService {
  constructor(
    private prisma: PrismaService,
    private jwtService: JwtService,
  ) {}

  async login(loginDto: LoginDto) {
    const user = await this.prisma.user.findUnique({
      where: { username: loginDto.username },
    });

    if (!user) {
      throw new UnauthorizedException('Invalid credentials');
    }

    const isPasswordValid = await bcrypt.compare(
      loginDto.password,
      user.password,
    );

    if (!isPasswordValid) {
      throw new UnauthorizedException('Invalid credentials');
    }

    if (!user.isActive) {
      throw new UnauthorizedException('Account is inactive');
    }

    // Update last login
    await this.prisma.user.update({
      where: { id: user.id },
      data: { lastLogin: new Date() },
    });

    const payload = {
      sub: user.id,
      username: user.username,
      role: user.role,
    };

    return {
      access_token: this.jwtService.sign(payload),
      user: {
        id: user.id,
        username: user.username,
        name: user.name,
        email: user.email,
        role: user.role,
      },
    };
  }

  async validateUser(userId: string) {
    return this.prisma.user.findUnique({
      where: { id: userId },
      select: {
        id: true,
        username: true,
        name: true,
        email: true,
        role: true,
        isActive: true,
      },
    });
  }
}

// src/modules/patients/patients.service.ts
import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { CreatePatientDto } from './dto/create-patient.dto';

@Injectable()
export class PatientsService {
  constructor(private prisma: PrismaService) {}

  async create(createPatientDto: CreatePatientDto) {
    // Generate unique patient ID
    const lastPatient = await this.prisma.patient.findFirst({
      orderBy: { createdAt: 'desc' },
      select: { patientId: true },
    });

    const nextNumber = lastPatient
      ? parseInt(lastPatient.patientId.split('-')[1]) + 1
      : 1;

    const patientId = `PID-${nextNumber.toString().padStart(4, '0')}`;

    return this.prisma.patient.create({
      data: {
        ...createPatientDto,
        patientId,
      },
    });
  }

  async findAll(page: number = 1, limit: number = 10) {
    const skip = (page - 1) * limit;

    const [patients, total] = await Promise.all([
      this.prisma.patient.findMany({
        skip,
        take: limit,
        orderBy: { createdAt: 'desc' },
      }),
      this.prisma.patient.count(),
    ]);

    return {
      data: patients,
      meta: {
        total,
        page,
        limit,
        totalPages: Math.ceil(total / limit),
      },
    };
  }

  async findOne(id: string) {
    const patient = await this.prisma.patient.findUnique({
      where: { id },
      include: {
        opdVisits: {
          include: { doctor: true },
          orderBy: { visitDate: 'desc' },
          take: 5,
        },
        admissions: {
          include: { doctor: true, bed: true },
          orderBy: { admissionDate: 'desc' },
          take: 5,
        },
        prescriptions: {
          include: { doctor: true },
          orderBy: { prescriptionDate: 'desc' },
          take: 5,
        },
      },
    });

    if (!patient) {
      throw new NotFoundException('Patient not found');
    }

    return patient;
  }

  async search(query: string) {
    return this.prisma.patient.findMany({
      where: {
        OR: [
          { patientId: { contains: query, mode: 'insensitive' } },
          { name: { contains: query, mode: 'insensitive' } },
          { phone: { contains: query } },
        ],
      },
      take: 10,
    });
  }
}

import { IsString, IsEnum, IsArray, IsNumber, IsOptional, ValidateNested } from 'class-validator';
import { Type } from 'class-transformer';
import { ApiProperty } from '@nestjs/swagger';
import { InvoiceType } from '@prisma/client';

class InvoiceItemDto {
  @ApiProperty()
  @IsString()
  description: string;

  @ApiProperty()
  @IsNumber()
  quantity: number;

  @ApiProperty()
  @IsNumber()
  unitPrice: number;
}

export class CreateInvoiceDto {
  @ApiProperty()
  @IsString()
  patientId: string;

  @ApiProperty({ enum: InvoiceType })
  @IsEnum(InvoiceType)
  invoiceType: InvoiceType;

  @ApiProperty({ type: [InvoiceItemDto] })
  @IsArray()
  @ValidateNested({ each: true })
  @Type(() => InvoiceItemDto)
  items: InvoiceItemDto[];

  @ApiProperty({ required: false })
  @IsNumber()
  @IsOptional()
  discount?: number;

  @ApiProperty({ required: false })
  @IsString()
  @IsOptional()
  notes?: string;
}

// src/modules/billing/billing.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { CreateInvoiceDto } from './dto/create-invoice.dto';
import { InvoiceStatus } from '@prisma/client';

@Injectable()
export class BillingService {
  constructor(private prisma: PrismaService) {}

  async createInvoice(dto: CreateInvoiceDto, userId: string) {
    // Generate invoice ID
    const lastInvoice = await this.prisma.invoice.findFirst({
      orderBy: { createdAt: 'desc' },
      select: { invoiceId: true },
    });

    const nextNumber = lastInvoice
      ? parseInt(lastInvoice.invoiceId.split('-')[1]) + 1
      : 1;

    const invoiceId = `INV-${nextNumber.toString().padStart(4, '0')}`;

    // Calculate totals
    const totalAmount = dto.items.reduce(
      (sum, item) => sum + item.quantity * item.unitPrice,
      0,
    );
    const discount = dto.discount || 0;
    const netAmount = totalAmount - discount;

    return this.prisma.invoice.create({
      data: {
        invoiceId,
        patientId: dto.patientId,
        invoiceType: dto.invoiceType,
        totalAmount,
        discount,
        netAmount,
        dueAmount: netAmount,
        status: InvoiceStatus.PENDING,
        createdById: userId,
        notes: dto.notes,
        items: {
          create: dto.items.map((item) => ({
            description: item.description,
            quantity: item.quantity,
            unitPrice: item.unitPrice,
            totalPrice: item.quantity * item.unitPrice,
          })),
        },
      },
      include: {
        items: true,
        patient: true,
        createdBy: {
          select: {
            id: true,
            name: true,
            role: true,
          },
        },
      },
    });
  }

  async addPayment(invoiceId: string, amount: number, method: string, userId: string) {
    const invoice = await this.prisma.invoice.findUnique({
      where: { id: invoiceId },
    });

    if (!invoice) {
      throw new NotFoundException('Invoice not found');
    }

    // Generate payment ID
    const lastPayment = await this.prisma.payment.findFirst({
      orderBy: { createdAt: 'desc' },
      select: { paymentId: true },
    });

    const nextNumber = lastPayment
      ? parseInt(lastPayment.paymentId.split('-')[1]) + 1
      : 1;

    const paymentId = `PAY-${nextNumber.toString().padStart(4, '0')}`;

    // Calculate new due amount
    const newPaidAmount = Number(invoice.paidAmount) + amount;
    const newDueAmount = Number(invoice.netAmount) - newPaidAmount;

    // Create payment
    const payment = await this.prisma.payment.create({
      data: {
        paymentId,
        invoiceId,
        amount,
        paymentMethod: method as any,
        receivedById: userId,
      },
    });

    // Update invoice
    await this.prisma.invoice.update({
      where: { id: invoiceId },
      data: {
        paidAmount: newPaidAmount,
        dueAmount: newDueAmount,
        status:
          newDueAmount === 0
            ? InvoiceStatus.PAID
            : InvoiceStatus.PARTIAL,
      },
    });

    return payment;
  }

  async getDues(page: number = 1, limit: number = 10) {
    const skip = (page - 1) * limit;

    const [invoices, total] = await Promise.all([
      this.prisma.invoice.findMany({
        where: {
          dueAmount: { gt: 0 },
        },
        include: {
          patient: true,
        },
        orderBy: { invoiceDate: 'desc' },
        skip,
        take: limit,
      }),
      this.prisma.invoice.count({
        where: {
          dueAmount: { gt: 0 },
        },
      }),
    ]);

    return {
      data: invoices,
      meta: {
        total,
        page,
        limit,
        totalPages: Math.ceil(total / limit),
      },
    };
  }
}

// src/lib/api.ts
import axios from 'axios';
import toast from 'react-hot-toast';

export const api = axios.create({
  baseURL: import.meta.env.VITE_API_URL || 'http://localhost:3000',
  headers: {
    'Content-Type': 'application/json',
  },
});

// Request interceptor
api.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem('token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// Response interceptor
api.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem('token');
      window.location.href = '/login';
      toast.error('Session expired. Please login again.');
    } else if (error.response?.status === 403) {
      toast.error('You do not have permission to perform this action');
    } else if (error.response?.data?.message) {
      toast.error(error.response.data.message);
    } else {
      toast.error('An error occurred. Please try again.');
    }
    return Promise.reject(error);
  }
);

// src/features/auth/api/useAuth.ts
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { api } from '@/lib/api';
import { useNavigate } from 'react-router-dom';
import toast from 'react-hot-toast';

interface LoginCredentials {
  username: string;
  password: string;
}

interface User {
  id: string;
  username: string;
  name: string;
  email: string;
  role: string;
}

export const useAuth = () => {
  const navigate = useNavigate();
  const queryClient = useQueryClient();

  const loginMutation = useMutation({
    mutationFn: async (credentials: LoginCredentials) => {
      const { data } = await api.post('/api/v1/auth/login', credentials);
      return data;
    },
    onSuccess: (data) => {
      localStorage.setItem('token', data.access_token);
      queryClient.setQueryData(['user'], data.user);
      toast.success('Login successful!');
      navigate('/dashboard');
    },
    onError: () => {
      toast.error('Invalid username or password');
    },
  });

  const { data: user, isLoading } = useQuery<User>({
    queryKey: ['user'],
    queryFn: async () => {
      const { data } = await api.post('/api/v1/auth/me');
      return data;
    },
    enabled: !!localStorage.getItem('token'),
    retry: false,
  });

  const logout = () => {
    localStorage.removeItem('token');
    queryClient.clear();
    navigate('/login');
    toast.success('Logged out successfully');
  };

  return {
    login: loginMutation.mutate,
    isLoggingIn: loginMutation.isPending,
    user,
    isLoadingUser: isLoading,
    logout,
  };
};

// src/features/patients/api/usePatients.ts
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { api } from '@/lib/api';
import toast from 'react-hot-toast';

export const usePatients = (page: number = 1, limit: number = 10) => {
  const queryClient = useQueryClient();

  const { data, isLoading } = useQuery({
    queryKey: ['patients', page, limit],
    queryFn: async () => {
      const { data } = await api.get(`/api/v1/patients?page=${page}&limit=${limit}`);
      return data;
    },
  });

  const createPatient = useMutation({
    mutationFn: async (patientData: any) => {
      const { data } = await api.post('/api/v1/patients', patientData);
      return data;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['patients'] });
      toast.success('Patient created successfully!');
    },
  });

  const searchPatients = useMutation({
    mutationFn: async (query: string) => {
      const { data } = await api.get(`/api/v1/patients/search?q=${query}`);
      return data;
    },
  });

  return {
    patients: data?.data || [],
    meta: data?.meta,
    isLoading,
    createPatient: createPatient.mutate,
    isCreating: createPatient.isPending,
    searchPatients: searchPatients.mutate,
    searchResults: searchPatients.data,
  };
};

   □ Register drhaiatgh.com
   □ Point to server IP
   □ Install SSL certificate (Let's Encrypt)
   □ Configure subdomains (console.drhaiatgh.com, api.drhaiatgh.com)

2. Server Setup
   □ Install Node.js 20 LTS
   □ Install PostgreSQL 15
   □ Install Redis 7
   □ Install Nginx
   □ Install PM2

3. Database Setup
   □ Create PostgreSQL database
   □ Run migrations: npx prisma migrate deploy
   □ Run seeders: npx prisma db seed

4. Backend Deployment
   □ Clone repository
   □ Install dependencies: npm install
   □ Build: npm run build
   □ Set up environment variables
   □ Start with PM2: pm2 start dist/main.js

5. Frontend Deployment
   □ Build: npm run build
   □ Copy dist folder to web server

6. Nginx Configuration
   □ Configure reverse proxy for API
   □ Configure static file serving for frontend
   □ Enable gzip compression

7. Security
   □ Configure firewall
   □ Set up database backups
   □ Configure CORS properly
   □ Enable rate limiting

8. Testing
   □ Test all API endpoints
   □ Test user roles and permissions
   □ Test invoice generation and printing
   □ Test SMS notifications
